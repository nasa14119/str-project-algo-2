CLASS Transmission
    ATTRIBUTES: name, content

    CONSTRUCTOR(fileName, path)
        name = fileName
        content = ReadFile(path)
    END CONSTRUCTOR


    METHOD SearchCode(mcode)
        (isFound, position) = Z_Function(content, mcode.content)

        IF isFound
            OUT code mcode.name, transimission and position
        ELSE
            OUT the code was not found
        END IF
    END METHOD


    METHOD SearchPalindrome()
        (start, length) ← Manacher(content)

        OUT `(start + 1) (inicio + length)`
    FIN METHOD


    METHOD SearchCommonSubstring(otherTranssmision)
        if content = otherTranssmision.content
            OUT this content
            RETURN METHOD
        END IF

        length = content.size() - 1

        WHILE length > 0

            FOR start = 0, start < content.size() - length
                pattern = extract substring(content, start to length)

                (isFound, position) =
                    Z_Function(otraTransmission.content, pattern)

                IF isFound
                    OUT pattern and position index + 1
                    RETURN METHOD
                END IF
            END FOR

            length = length - 1
        END WHILE
    END METHOD
FIN CLASE
